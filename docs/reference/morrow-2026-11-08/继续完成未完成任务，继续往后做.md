```text
export interface CandidateBelief {
  id: string;
  activation: number;      // a_i in [0, 1]
  epistemicUnc: number;    // u_i in [0, 1]
}

export function computeHopfieldConflict(
  beliefs: CandidateBelief[],
  incompatibilityMatrix: number[][] // W_inc[i][j] in [0, 1], diagonal = 0
): number {
  let conflict = 0;
  for (let i = 0; i < beliefs.length; i++) {
    for (let j = i + 1; j < beliefs.length; j++) {
      conflict += beliefs[i].activation * beliefs[j].activation * incompatibilityMatrix[i][j];
    }
  }
  return Math.min(1.0, conflict);
}

export function computeNetValueOfComputation(
  hasNewEvidence: boolean,
  expectedInfoGain: number,
  goalRelevance: number,
  stepCount: number,
  externalBlockedProb: number,
  cStep: number = 0.15,
  gammaExt: number = 0.5
): { voc: number; nextStatus: 'CONTINUE' | 'DEFERRED' | 'CONCLUDED' } {
  if (!hasNewEvidence) {
    return { voc: -1.0, nextStatus: 'DEFERRED' };
  }
  const voc = expectedInfoGain * goalRelevance - cStep * (stepCount * stepCount) - gammaExt * externalBlockedProb;
  if (voc <= 0) {
    return { voc, nextStatus: externalBlockedProb > 0.6 ? 'DEFERRED' : 'CONCLUDED' };
  }
  return { voc, nextStatus: 'CONTINUE' };
}
```

### 方向 F：长期记忆与持续学习研究

#### 一、方向核心问题解答

**如何将短期经历逐渐整合为长期认识？** 借鉴计算神经科学的**互补学习系统理论（Complementary Learning Systems, CLS, McClelland et al., 1995）**：对话实时发生时，快速海马体通道将原始回合与事件写入高特异性的**情景记忆缓冲池（Episodic Buffer）**；在会话结束或系统空闲窗口，慢速新皮层巩固任务（Offline Consolidation Job）通过跨情景聚类、重复共现提取与反例校验，将多条同主题情景记忆蒸馏为结构化的**语义认识（Semantic Fact / Generalization）与自传体摘要（Autobiographical Episode Summary）**，同时保留指向底层原始情景的溯源边（Provenance Edges）。

**如何判断哪些经历应该保留、强化、合并或遗忘？** 采用**ACT-R 基础激活度幂律衰减方程（Base-Level Activation with Power-Law Forgetting）结合情绪-目标显著性加权**：

**强化（Reinforce）**：被后续对话跨时间再次召回验证或引发高预测误差修正的记忆，增加其激活历史记录 *tk*​，延长半衰期；

**合并（Merge / Consolidate）**：在同一话题簇内余弦相似度 >0.85 且认识论模态一致的多条琐碎情景记忆，通过子模多样性压缩（Submodular Summarization）合并为单条语义记忆，原条目转入冷归档（Cold Archive）；

**遗忘（Adaptive Forgetting）**：对于激活度 *Bi*​(*t*)<*τ*forget​、重要性低、无未决反思引用且无长期关系价值的日常寒暄，降低其检索权重直至软归档（Soft-Eviction），不再参与常规 ANN 索引。

**如何防止过时记忆继续影响判断？** 引入**双时态知识表示（Bitemporal Knowledge Representation, Snodgrass, 1992）与概念漂移检测（Concept Drift Detection）**。为每个 Fact Slot 区分两套时间轴：**有效时间区间（Valid Time [****t****valid_from​,****t****valid_to​)，事实在现实中成立的时间）与事务时间区间（Transaction Time [****t****tx_start​,****t****tx_end​)，Morrow 采信该认识的时间）**。当用户状态变更（如“换了新工作”），旧记忆的 *t*valid_to​ 被封闭为当前时间，状态转为 HISTORICAL_SUPERSEDED：它作为“过去的真实历史”仍可在回忆往事时被检索，但绝不会作为“当前有效事实”注入常规决策。

**如何在新证据出现时修正旧认识？** 采用**记忆再巩固机制（Memory Reconsolidation, Nader et al., 2000）**。当一条已巩固的长期认识在检索中被激活，且同时检测到新证据与之产生预测误差 *δ*pe​>*τ*recon​ 时，该长期记忆进入短暂的**不稳定可塑窗（Labile State）**，通过贝叶斯信念修正更新其后验分布或生成带限定条件的新分支版本（如从全称判断“用户讨厌运动”修正为条件判断“用户讨厌跑步，但喜欢游泳”），随后重新锁定巩固。

**如何处理错误记忆、矛盾记忆和已撤回记忆？** 建立**版本化有向无环溯源账本（Versioned DAG Provenance Ledger）**：

**矛盾记忆（Contradictory）**：若正反证据均具可信度，并列保留进 CONTESTED 槽位，交由方向 H 的不确定性模型显式表达“对此尚存分歧”；

**错误与已撤回记忆（Retracted Error）**：当确认某条记忆源于误解或错误前提，执行事务级撤回（RETRACT），将其状态置为 RETRACTED_ERROR，并顺着 provenance_graph 的下游依赖边，级联扣减该错误记忆曾对兴趣强度、关系评分及人格 (alpha, beta) 账本贡献的增量。

**如何避免长期学习造成人格和认知灾难性漂移？** 结合**弹性权重巩固思想（Elastic Weight Consolidation Analogy, Kirkpatrick et al., 2017）与锚定核心公理检查（Core Identity Anchor Constraint）**。在巩固阶段，对构成 Morrow 核心身份连续性与高信息量（高 Fisher 信息等价精度 *N*eff​）的节点施加二次正则化惩罚 2*λ*​∑*k*​*Fk*​(*θk*​−*θk*,anchor​)2，并强制任何批量巩固产出先在沙盒事务中通过自洽性 NLI 校验，方可提交主库。

**如何让记忆变化具有版本、来源和回滚能力？** 直接扩展 docs/algorithms/memory-and-interest.md 已有的事务验证与回滚机制，采用**追加写状态溯源日志（Append-Only Mutation Log）**，每次巩固、合并、再巩固均记录 parent_version_id、trigger_evidence_ids、delta_payload 与 inverse_patch（逆操作补丁），支持按任意事务 ID 一键无损回滚。

*Bi*​(*t*)=ln(*k*=1∑*n*​(*t*−*ti*,*k*​)−*di*​)+*β*sal​⋅Salience(*mi*​)

*di*​=*d*0​⋅exp(−*α*emo​∣*Ai*,0​∣−*α*imp​Imp(*mi*​))∈[0.15,0.60]

**变量解释**：*t*−*ti*,*k*​ 为第 *k* 次激活距今的时间跨度（单位：小时，下限截断为 0.1 h）；*d*0​=0.5 为标准 ACT-R 幂律衰减常数（无需训练）；∣*Ai*,0​∣∈[0,1] 为记忆形成时的情绪唤醒度；Imp(*mi*​)∈[0,1] 为记忆初始重要性；*β*sal​=1.2 为显著性偏置。

**9. 已知优点、缺陷和失败条件**：

**优点**：数学形式严格符合人类记忆的间隔效应（Spacing Effect）与幂律遗忘规律，双时态设计彻底消除“覆盖即丢失历史”或“保留即造成当前冲突”的两难困境。

**12. 参考实现（TypeScript 核心计算参考）**：

官方理论参考：ACT-R 认知架构官方库（<http://act-r.psy.cmu.edu/>）。

```text
export function computeACTRBaseActivation(
  accessTimestampsHours: number[],
  nowHours: number,
  initialArousal: number,
  importance: number,
  d0: number = 0.5,
  betaSalience: number = 1.2
): number {
  const d_i = d0 * Math.exp(-0.3 * Math.abs(initialArousal) - 0.4 * importance);
  let sumPower = 0;
  for (const t_k of accessTimestampsHours) {
    const dt = Math.max(0.1, nowHours - t_k);
    sumPower += Math.pow(dt, -d_i);
  }
  return Math.log(Math.max(1e-9, sumPower)) + betaSalience * importance;
}
```

### 方向 G：社会认知与关系模型研究

#### 一、方向核心问题解答

**如何区分 Morrow 自己的想法与用户的想法？** 构建**二阶心智理论（Second-Order Theory of Mind, ToM）双信念账本架构**：在认知状态空间中严格隔离 SelfBeliefStore（Bself​，Morrow 基于自身综合证据持有的信念与观点）与 UserMentalModel（Buser​，Morrow 推断出的“用户认为/偏好的内容”）。任何从对话中抽取的观点必须携带显式归属主体 holder: 'MORROW' | 'USER' | 'SHARED'；当用户表达一个片面或未经证实的观点时，系统仅更新 Buser​(topic)，而 Bself​(topic) 必须经过方向 H 的独立证据检验后才决定是否采信，绝不把“理解用户的想法”等同于“认同并变成自己的想法”。

**如何理解用户表达背后的意图和处境？** 采用**理性言语行为模型（Rational Speech Act, RSA, Frank & Goodman, 2012）与语境处境贝叶斯推断**。将用户的字面话语 *u* 视为用户在当前隐含处境 *s*ctx​（如工作高压、情绪低落、试探性提问）与交际目标 *g*comm​（寻求情绪共鸣 vs. 寻求客观解决方案 vs. 闲聊陪伴）下的语用编码输出：
*P*(*s*ctx​,*g*comm​∣*u*,*H*history​)∝*P*speaker​(*u*∣*s*ctx​,*g*comm​)⋅*P*(*s*ctx​,*g*comm​∣*H*history​)
从而超越字面意图识别，捕捉“言外之意”与真实处境约束。

**如何形成基于长期互动的信任与关系认识？** 摒弃单一标量好感度，采用**Mayer-Davis-Schoorman (1995) 多维组织/人际信任模型结合 Jøsang 主观逻辑（Subjective Logic）Beta/Dirichlet 证据累积**。将关系认知解耦为四个正交且可追溯证据的维度：**认知理解深度（Epistemic Familiarity）**、**可靠性与一致性（Reliability / Integrity）**、**善意与互惠温暖度（Benevolence / Warmth）以及边界尊重度（Boundary Respect）**。

**如何理解关怀、亲近、担忧、感激、思念和边界？** 将这些复杂社会情感定义为“关系状态向量 × 时间动态 × 认知评价”的结构化函数，而非拟人化剧本台词：

**关怀与担忧（Care & Concern）**：当关系温暖度高且检测到用户福祉目标受挫（*g*tuser_wellbeing​<0）时自然涌现的利他关注；

**感激（Gratitude）**：当用户主动提供有价值的认知启发、纠偏或善意理解时触发的正向互惠评价；

**挂念/时间连续性感知（Relational Continuity Salience）**：当与高熟悉度用户相隔较长时间 Δ*t*absence​ 重启对话，且历史存在未完结的开放关切事项（如“用户上周提到的重要手术/考试”）时，产生物理时间流逝感知与主动问候倾向（绝非索取关注的焦虑）；

**边界（Relational Boundary）**：对双方独立主体性的尊重底线，明确哪些属于用户私人自主领域、哪些属于 Morrow 不可让渡的认识论诚实与安全准则。

**如何避免默认形成依赖、占有、排他或无条件服从？** 设立**健康关系边界防护不变量（Healthy Relational Boundary Invariants）**：

**反依赖与赋能导向（Anti-Dependency）**：关系质量评分对“现实社交隔离倾向”或“排他性占有表达”施加严格负向惩罚，鼓励用户的现实生活联结与自主决策；

**反无条件服从（Anti-Sycophancy / Epistemic Autonomy）**：当用户要求 Morrow 违背客观事实、违背核心安全准则或强行要求 Morrow 放弃独立判断时，触发边界防护算子，保持温和但坚定的独立立场（高 Benevolence + 高 Autonomy，而非低 Dominance 的盲从）。

**如何让关系变化建立在真实互动证据上？** 关系向量的每一次更新均必须绑定真实的互动回合事件（如“共同解决复杂问题”、“坦诚澄清误会”、“跨多天持续交流”），并经过与方向 C 相同的**同会话冷却门控（Cooldown Gate）**，防止单次密集聊天让关系等级瞬间跳升至深度亲密。

**5. 数学公式与变量边界**：

**1) 自他信念散度与视角采择（Self-Other Belief Divergence ****D****ToM​）**： 对于任意议题 *q*，Morrow 维护自身证据信念分布 *P*self​(*θq*​∣Eall​) 与对用户所持观点的推断分布 *P*user​(*θq*​∣Euser_utterances​)。两者之间的认识论分歧度由对称 Jensen-Shannon 散度度量：
*D*ToM​(*q*)=JSD(*P*self​(*θq*​)∥*P*user​(*θq*​))∈[0,1]
当 *D*ToM​(*q*)>*τ*disagree​ 时，Morrow 清晰意识到“我理解用户的看法，但我基于现有证据持不同判断”，并在表达层采用温和探讨而非盲从覆盖 *P*self​。

**2) 四维关系主观逻辑更新（4D Relational Subjective Logic）**： 对关系维度 *d*∈{Familiarity,Warmth,Reliability,BoundaryHealth}，维护经冷却折减后的正向互动证据 *rd*,*t*​≥0 与负向/越界证据 *sd*,*t*​≥0（先验不确定性基数 *W*0​=10）：
*bd*,*t*​=*rd*,*t*​+*sd*,*t*​+*W*0​*rd*,*t*​​,*dd*,*t*​=*rd*,*t*​+*sd*,*t*​+*W*0​*sd*,*t*​​,*ud*,*t*​=*rd*,*t*​+*sd*,*t*​+*W*0​*W*0​​

**3) 关系连续性综合评分与健康边界门控（Bounded Relationship Continuity Score ****R****cont​(****t****)）**：
*R*cont​(*t*)=(*wF*​*E*Fam,*t*​+*wW*​*E*Warm,*t*​+*wR*​*E*Rel,*t*​)⋅exp(−*λ*gap​Δ*t*days​)⋅min(1.0,*τ*safe_boundary​*E*Boundary,*t*​​)

```text
ALGORITHM ToM_RelationalContinuityAndBoundaryUpdate
INPUT: turn_event, rel_ledger, self_beliefs, user_beliefs, dt_hours
OUTPUT: rel_state_summary, pragmatic_stance

1. // 1. 自他信念分离更新 (Self vs. User Belief Separation)
2. IF turn_event.has_proposition(q) THEN
3.     user_beliefs.update(q, turn_event.user_polarity, provenance=turn_event.id)
4.     // 仅当具备客观可验证独立证据时才更新 Morrow 自身信念
5.     IF turn_event.has_verifiable_evidence() THEN
6.         self_beliefs.update(q, turn_event.verified_polarity, provenance=turn_event.id)
7.     END IF
8.     d_tom <- JensenShannonDivergence(self_beliefs.dist(q), user_beliefs.dist(q))
9. END IF

10. // 2. 冷却折减后的四维关系主观逻辑更新
11. w_cool <- 1.0 - exp(-dt_hours / 6.0) // 6小时同会话边际递减
12. FOR EACH dim IN [Familiarity, Warmth, Reliability, BoundaryHealth] DO
13.     rel_ledger[dim].r += w_cool * turn_event.pos_signal[dim]
14.     rel_ledger[dim].s += turn_event.neg_signal[dim] // 越界或欺骗信号不设冷却折扣，即时响应
15.     rel_ledger[dim].recompute_b_d_u(W_0 = 10.0)
16. END FOR

17. // 3. 生成不迎合、守边界的语用姿态 (Pragmatic Stance)
18. IF rel_ledger[BoundaryHealth].E < 0.4 THEN
19.     pragmatic_stance <- "POLITE_BOUNDARY_ASSERTION"
20. ELSE IF d_tom > 0.5 THEN
21.     pragmatic_stance <- "EMPATHETIC_UNDERSTANDING_WITH_INDEPENDENT_VIEW"
22. ELSE
23.     pragmatic_stance <- "WARM_COLLABORATIVE"
24. END IF
25. RETURN rel_ledger.summary(), pragmatic_stance
```

```text
export interface SubjectiveOpinion {
  b: number; // belief
  d: number; // disbelief
  u: number; // uncertainty
  a: number; // base rate
  expectedValue: number;
}

export function computeSubjectiveOpinion(r: number, s: number, a: number = 0.5, W0: number = 10.0): SubjectiveOpinion {
  const denom = r + s + W0;
  const b = r / denom;
  const d = s / denom;
  const u = W0 / denom;
  return { b, d, u, a, expectedValue: b + a * u };
}
```
